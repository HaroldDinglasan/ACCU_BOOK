CREATE TABLE SystemUser (
    Oid UNIQUEIDENTIFIER NOT NULL DEFAULT NEWID(),  -- Primary Key
    StoredPassword NVARCHAR(100) NOT NULL,          -- Password string
    ChangePasswordOnFirstLogon BIT NOT NULL,        -- 1 or 0 (true/false)
    UserName NVARCHAR(50) NOT NULL UNIQUE,          -- Username (unique)
    IsActive BIT NOT NULL DEFAULT 1,                -- Active status
    OptimisticLockField INT NOT NULL DEFAULT 0,     -- Concurrency field
    FullName NVARCHAR(150) NOT NULL,                -- User’s full name
    Remarks NVARCHAR(MAX) NULL,                     -- Optional remarks
    Picture VARBINARY(MAX) NULL,                    -- Store image (if needed)
    Signature VARBINARY(MAX) NULL,                  -- Digital signature (optional)
    Email NVARCHAR(150) NULL,                       -- User email
    RoleType NVARCHAR(50) NOT NULL                  -- Role (e.g., Admin, User)
);
