public class Token
{
    private readonly IConfiguration _config;
    private readonly IUserData userData;
   // private UserModel userForAuthentication { get; set; }
    public Token(IConfiguration configuration, IUserData userData)
    {
        _config = configuration;
        this.userData = userData;
    }

    public async Task<TokenResponse> PutToken(UserModel userForAuthentication)
    {
        // Isang beses lang: valid ang user at password kung may ibinalik na account
        var account = await userData.LoginUser(userForAuthentication);

        if (account == null)
        {
            return new TokenResponse { Success = false };
        }

        var token = GenerateToken(account);

        return new TokenResponse
        {
            Success = true,
            Access_Token = token.Access_Token,
            UserName = token.UserName
        };
    }
 



    private (string Access_Token, string UserName) GenerateToken(AuthenticatedUserAccount account)
    {
        var claims = new List<Claim>
        {
            new Claim(ClaimTypes.Name, account.UserName),
            new Claim(ClaimTypes.GivenName, account.FullName ?? string.Empty),
            new Claim(JwtRegisteredClaimNames.Nbf, new DateTimeOffset(DateTime.Now).ToUnixTimeSeconds().ToString()),
            new Claim(JwtRegisteredClaimNames.Exp, new DateTimeOffset(DateTime.Now.AddDays(1)).ToUnixTimeSeconds().ToString()),
            new Claim(ClaimTypes.Role, account.RoleType ?? string.Empty),
            new Claim(ClaimTypes.NameIdentifier, account.Oid.ToString()),
            new Claim("AccessName", account.RoleName ?? string.Empty),
            new Claim("RoleId", account.RoleId.ToString()),
            new Claim("ChangePasswordOnFirstLogon", account.ChangePasswordOnFirstLogon.ToString())
        };

        var token = new JwtSecurityToken(
            new JwtHeader(
                new SigningCredentials(
                    new SymmetricSecurityKey(Encoding.UTF8.GetBytes(_config["Jwt:Key"])),
                    SecurityAlgorithms.HmacSha256)),
            new JwtPayload(claims));

        return (new JwtSecurityTokenHandler().WriteToken(token), account.UserName);
    }

 public async Task<AuthenticatedUserAccount> LoginUser(UserModel userModel)
 {
     if (userModel == null ||
         string.IsNullOrWhiteSpace(userModel.UserName) ||
         string.IsNullOrEmpty(userModel.StoredPassword))
         return null;

     // Username lang ang sa WHERE. Wala na ang StoredPassword filter.
     string sql = @"SELECT SU.Oid,
                           StoredPassword,
                           ChangePasswordOnFirstLogon,
                           UserName,
                           IsActive,
                           FullName,
                           Remarks,
                           Picture,
                           Signature,
                           Email,
                           RoleType,
                           SR.Name AS RoleName,
                           SSR.RolesId AS RoleId
                    FROM SystemUser SU
                    LEFT JOIN SystemUser_SystemRoles SSR ON SSR.Users = SU.Oid
                    LEFT JOIN SystemRole SR ON SR.Oid = SSR.RolesId
                    WHERE SU.IsActive = 1 AND UserName = @UserName";

     var model = await db.LoadSingleData<AuthenticatedUserAccount>(
         sql, new
         {
             userModel.UserName
         });

     if (model == null)
         return null;

     // I-verify ang plain password laban sa hash sa DB
     if (!_passwords.Verify(userModel.StoredPassword, model.StoredPassword))
         return null;

     // Huwag nang dalhin ang hash sa ibang parte ng app
     model.StoredPassword = null;

     return model;
 }

}

namespace Accounting_V4.Services.AuthenServices;

public class AuthenticationService : IAuthenticationService
{
    private readonly HttpClient _client;
    private readonly AuthenticationStateProvider _authenticationStateProvider;
    private readonly ILocalStorageService _localStorage;
    private JsonSerializerOptions _options = new() { PropertyNameCaseInsensitive = true };
    private Token _token;
    public AuthenticationService(Token token, HttpClient client,
        AuthenticationStateProvider authenticationStateProvider,
        ILocalStorageService localStorage)
    {
        _token = token;
        _client = client;
        _authenticationStateProvider = authenticationStateProvider;
        _localStorage = localStorage;
    }

    public async Task<AuthenticatedUserModel> Login(UserModel userForAuthentication)
    {
        var authResult = await _token.PutToken(userForAuthentication);

        if (authResult != null && authResult.Success)
        {
            var result = new AuthenticatedUserModel
            {
                Access_Token = authResult.Access_Token,
                UserName = authResult.UserName
            };

            await _localStorage.SetItemAsync("tokenStorage", authResult.Access_Token);

            ((CustomAuthStateProvider)_authenticationStateProvider)
                .NotifyUserAuthentication(authResult.Access_Token);

            _client.DefaultRequestHeaders.Authorization =
                new AuthenticationHeaderValue("bearer", authResult.Access_Token);

            return result;
        }

        // failed login
        return new AuthenticatedUserModel();
    }

    public async Task LogOut()
    {
        await _localStorage.RemoveItemAsync("tokenStorage");
        ((CustomAuthStateProvider)_authenticationStateProvider).NotifyUserLogOut();
        _client.DefaultRequestHeaders.Authorization = null;
    }
}


using System.Reflection;

namespace Accounting_V4.Services.Security;


public interface IDevExpressPasswordService
{
    string Hash(string plainPassword);
    bool Verify(string plainPassword, string storedHash);
}
public class DevExpressPasswordService : IDevExpressPasswordService
{
    private readonly Type cryptographerType;
    private readonly MethodInfo hashMethod;
    private readonly MethodInfo verifyMethod;

    public DevExpressPasswordService()
    {
        string path = Path.Combine(AppContext.BaseDirectory, "DevExpress", "DevExpress.Persistent.Base.v20.1.dll");
        if (!File.Exists(path))
            path = Path.Combine(AppContext.BaseDirectory, "DevExpress.Persistent.Base.v20.1.dll");
        if (!File.Exists(path))
            throw new FileNotFoundException("DevExpress password DLL was not found.");

        Assembly assembly = Assembly.LoadFrom(path);
        cryptographerType = assembly.GetType("DevExpress.Persistent.Base.PasswordCryptographer", true);

        const BindingFlags flags = BindingFlags.Static | BindingFlags.Public | BindingFlags.NonPublic;

        // Static ito, kaya target = null
        SetStatic("EnableRfc2898", false);
        SetStatic("SupportLegacySha512", true);

        hashMethod = cryptographerType.GetMethod("HashPassword", flags, null, new[] { typeof(string) }, null)
            ?? throw new MissingMethodException("DevExpress HashPassword method was not found.");

        verifyMethod = cryptographerType.GetMethod("VerifyHashedPassword", flags, null,
                           new[] { typeof(string), typeof(string) }, null)
            ?? throw new MissingMethodException("DevExpress VerifyHashedPassword method was not found.");
    }

    private void SetStatic(string name, object value)
    {
        const BindingFlags flags = BindingFlags.Static | BindingFlags.Public | BindingFlags.NonPublic;

        PropertyInfo prop = cryptographerType.GetProperty(name, flags);
        if (prop != null && prop.CanWrite)
        {
            prop.SetValue(null, value);
            return;
        }

        FieldInfo field = cryptographerType.GetField(name, flags);
        field?.SetValue(null, value);
    }

    public string Hash(string plainPassword)
    {
        if (string.IsNullOrEmpty(plainPassword))
            throw new ArgumentException("Password is required.", nameof(plainPassword));

        return (string)hashMethod.Invoke(null, new object[] { plainPassword });
    }

    public bool Verify(string plainPassword, string storedHash)
    {
        if (string.IsNullOrEmpty(plainPassword) || string.IsNullOrEmpty(storedHash))
            return false;

        try
        {
            // Order ng parameters: (hashedPassword, password)
            return (bool)verifyMethod.Invoke(null, new object[] { storedHash, plainPassword });
        }
        catch (TargetInvocationException)
        {
            return false; // corrupt/invalid hash format
        }
    }
}
